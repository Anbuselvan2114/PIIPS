    --nssm setup
        $ErrorActionPreference = "Stop"

        $AppRoot = "E:\Source\piips"
        $Python = "$AppRoot\venv\Scripts\python.exe"
        $Nssm = "$AppRoot\required_softwares\nssm-2.24\win64\nssm.exe"
        $ServiceName = "PIIPS_Backend"
        $LogDir = "$AppRoot\logs"

        Write-Host "=== PIIPS NSSM LIVE SETUP ===" -ForegroundColor Cyan

        # Check required files
        if (!(Test-Path $Python)) {
            throw "Python venv not found: $Python"
        }

        if (!(Test-Path $Nssm)) {
            throw "NSSM not found: $Nssm"
        }

        # Create log folder
        New-Item -ItemType Directory -Force -Path $LogDir | Out-Null

        # Install service if it does not exist
        & $Nssm status $ServiceName 2>$null

        if ($LASTEXITCODE -ne 0) {
            Write-Host "Installing NSSM service..." -ForegroundColor Yellow
            & $Nssm install $ServiceName $Python
        }

        # Configure service
        & $Nssm set $ServiceName Application $Python
        & $Nssm set $ServiceName AppDirectory $AppRoot
        & $Nssm set $ServiceName AppParameters "-m uvicorn main:app --host 127.0.0.1 --port 8000"

        # Automatic startup
        & $Nssm set $ServiceName Start SERVICE_AUTO_START

        # Logs
        & $Nssm set $ServiceName AppStdout "$LogDir\piips_stdout.log"
        & $Nssm set $ServiceName AppStderr "$LogDir\piips_stderr.log"

        # Restart service
        Write-Host "Starting PIIPS_Backend..." -ForegroundColor Yellow
        & $Nssm restart $ServiceName

        Start-Sleep -Seconds 3

        # Check service
        Get-Service $ServiceName

        Write-Host ""
        Write-Host "=== PIIPS BACKEND STATUS ===" -ForegroundColor Green
        & $Nssm status $ServiceName

        Write-Host ""
        Write-Host "=== CHECK PORT 8000 ===" -ForegroundColor Green
        netstat -ano | findstr :8000

        Write-Host ""
        Write-Host "PIIPS Backend configured successfully." -ForegroundColor Green
        Write-Host "Application Root : $AppRoot"
        Write-Host "Python           : $Python"
        Write-Host "Uvicorn          : 127.0.0.1:8000"
        Write-Host "IIS Domain       : https://piips.precisionit.co.in:8010"
